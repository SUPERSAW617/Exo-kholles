\newenvironment{exercise}[1][]
{
    \refstepcounter{exercise}
    \begin{tcolorbox}[
        breakable,
        title={Exercice \theexercise\ifx&#1&\else\ -- #1\fi},
        colback=white,
        colframe=black,
        boxrule=0.7pt,
        arc=2mm,
        left=3mm,
        right=3mm,
        top=2mm,
        bottom=2mm
    ]
}
{
    \end{tcolorbox}
}

}

}
